using Xunit;
using CompotClicker.Services;

namespace CompotClicker.Tests.ServicesTests
{
    public class GameServiceTests
    {
        [Fact]
        public void AddPoints_ShouldIncreaseTotalPoints()
        {
            // Arrange
            var gameService = new GameService();

            // Act
            gameService.AddPoints(10);

            // Assert
            Assert.Equal(10, gameService.TotalPoints);
        }
    }
}



[Fact]
public void LevelUp_ShouldIncreaseXPToNextLevel_AndSave()
{
    var gs = GameService.Instance;
    gs.ResetGame();

    double before = gs.XPToNextLevel;
    gs.GainExperience(10000); // wymuszenie level up

    Assert.True(gs.Level > 1);
    Assert.True(gs.XPToNextLevel > before);
}


[Fact]
public void XPToNextLevel_ShouldInitializeProperly()
{
    var gs = GameService.Instance;
    gs.ResetGame();

    Assert.True(gs.XPToNextLevel > 0);
}



[Fact]
public void Purchase_ShouldReduceAvailableXP_AndNotify()
{
    var gs = GameService.Instance;
    gs.ResetGame();
    gs.TryPurchaseXP(0, 0); // inicjalizacja

    gs.GainExperience(200); // zdobycie XP
    var item = new UpgradeItem("u1", "Test", 100, 1);

    bool notified = false;
    MessagingCenter.Subscribe<UpgradeItem>(this, "UpgradePurchased", _ => notified = true);

    item.BuyCommand.Execute(null);

    Assert.True(item.IsPurchased);
    Assert.True(notified);
    Assert.True(gs.AvailableExperience < 200);
}



[Fact]
public void ShopViewModel_TotalPoints_ShouldMatchAvailableXP()
{
    var gs = GameService.Instance;
    var vm = new ShopViewModel();

    gs.ResetGame();
    gs.TryPurchaseXP(0, 0); // wywołanie zapisu

    Assert.Equal(gs.AvailableExperience, vm.TotalPoints);
}



[Fact]
public void ShopViewModel_ShouldInitializeProperly()
{
    var vm = new ShopViewModel();
    Assert.NotNull(vm.Upgrades);
    Assert.True(vm.Upgrades.Count > 0);
}
